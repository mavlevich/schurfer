"""Planning input for HYP-030: power for 1-minute bursts on Bybit (not a test).

Dispersion scenario: the 732 frozen decay firings, gross from the t+1 bar open to the
t+61 bar open (ticker-based bar opens). Returns are centered by the planning code, so
only shape, tails and dependence matter. Inputs are verified (the firing list against its
sha256 sidecar) and pinned in the output with the code revision and the parameters.
"""
from datetime import UTC, datetime

import json
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[4]
sys.path.insert(0, str(ROOT / "apps" / "analytics"))
from schurfer_analytics import research_cost_power_planning as p  # noqa: E402
from schurfer_analytics.bybit_burst_decay import verified_bars_against_read2  # noqa: E402
from schurfer_analytics.mexc_early_trigger_hyp029 import verified_revision  # noqa: E402
from schurfer_analytics.source_lead_multi_source_report import load_verified  # noqa: E402

RUNTIME = Path("/Users/mavlevich/Desktop/Projects/schurfer/runtime/research")
FIRINGS = ROOT / "docs/research/evidence/bybit-burst-decay-v1/decay-firings.json"
PINNED = RUNTIME / "edge-loss/read-2/inputs.json"
BARS = RUNTIME / "edge-loss/bars"
head = subprocess.run(["git", "-C", str(ROOT), "rev-parse", "HEAD"], capture_output=True,
                      text=True, check=True).stdout.strip()
revision = verified_revision(head)  # a clean checkout of an explicit commit
payload, firings_sha = load_verified(FIRINGS)  # the frozen decay firings and their sidecar
firings = payload["firings"]

PARAMS = {"replicates": 1000, "calibration_replicates": 200, "calibration_iterations": 1000,
          "seed": 20261005, "return": "t61_open / t1_open - 1"}
episodes = [
    p.Episode(cluster=f["symbol"], at=datetime.fromtimestamp(f["bar_start"], UTC),
              net_bps=(f["t61_open"] / f["t1_open"] - 1) * 1e4)
    for f in firings
]
out = p.dataset_power(episodes, label="hyp030_bursts", replicates=PARAMS["replicates"],
                      calibration_replicates=PARAMS["calibration_replicates"],
                      calibration_iterations=PARAMS["calibration_iterations"],
                      seed=PARAMS["seed"])
days = len({e.at.date() for e in episodes})
out["flow"] = {"firings": len(episodes), "utc_days": days, "per_day": len(episodes) / days,
               "instruments": len({e.cluster for e in episodes})}
out["provenance"] = {"code_revision": revision, "decay_firings_sha256": firings_sha,
                     "parameters": PARAMS}
print(json.dumps(out, indent=1, default=str))
